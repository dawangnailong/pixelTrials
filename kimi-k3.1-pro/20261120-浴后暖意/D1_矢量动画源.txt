def f(v: float) -> str:
    return f'{v:.2f}'.rstrip('0').rstrip('.')
def clamp(x,a=0.,b=1.): return max(a,min(b,x))
def mix(a,b,u): return a+(b-a)*u
def smooth(x,a,b):
    u=clamp((x-a)/(b-a)); return u*u*(3-2*u)
def ease(x): return x*x*(3-2*x)
def color_mix(c1,c2,u):
    a=tuple(int(c1[i:i+2],16) for i in (1,3,5)); b=tuple(int(c2[i:i+2],16) for i in (1,3,5))
    return '#'+''.join(f'{round(mix(a[i],b[i],u)):02x}' for i in range(3))
def attrs(**kw):
    out=[]
    for k,v in kw.items():
        if v is None: continue
        k=k.rstrip('_').replace('_','-')
        out.append(f'{k}="{escape(str(v))}"')
    return ' '.join(out)
def el(name,content='',**kw): return f'<{name} {attrs(**kw)}>{content}</{name}>'
def empty(name,**kw): return f'<{name} {attrs(**kw)}/>'
def group(content,id_,**kw): return el('g',content,id=id_,**kw)
def path(d,**kw): return empty('path',d=d,**kw)
def ellipse(cx,cy,rx,ry,**kw): return empty('ellipse',cx=f(cx),cy=f(cy),rx=f(rx),ry=f(ry),**kw)
def circle(cx,cy,r,**kw): return empty('circle',cx=f(cx),cy=f(cy),r=f(r),**kw)
def line(x1,y1,x2,y2,**kw): return empty('line',x1=f(x1),y1=f(y1),x2=f(x2),y2=f(y2),**kw)
def rect(x,y,w,h,**kw): return empty('rect',x=f(x),y=f(y),width=f(w),height=f(h),**kw)
def txt(x,y,s,**kw): return el('text',escape(s),x=f(x),y=f(y),**kw)

def keys(t, table):
    if t<=table[0][0]: return table[0][1]
    for (ta,a),(tb,b) in zip(table,table[1:]):
        if t<=tb:
            u=ease((t-ta)/(tb-ta))
            if isinstance(a,(tuple,list)): return tuple(mix(x,y,u) for x,y in zip(a,b))
            return mix(a,b,u)
    return table[-1][1]

def steam(t,foreground=False):
    rng=random.Random(SEED+(91 if foreground else 17)); items=[]
    count=6 if foreground else 11
    for i in range(count):
        x0=rng.uniform(120,1800); y0=rng.uniform(460,990); amp=rng.uniform(18,58)
        speed=rng.uniform(12,28); phase=rng.uniform(0,math.tau)
        rise=(t*speed+i*53)%330
        y=y0-rise; x=x0+amp*math.sin(t*0.48+phase)
        op=(0.035 if foreground else 0.055)*(0.65+0.35*math.sin(math.pi*rise/330))
        d=f'M{f(x)} {f(y+145)} C{f(x-amp)} {f(y+100)} {f(x+amp)} {f(y+45)} {f(x)} {f(y)} C{f(x-amp*.7)} {f(y-45)} {f(x+amp*.6)} {f(y-80)} {f(x+5)} {f(y-120)}'
        items.append(path(d,fill='none',stroke=P['steam'],stroke_width=f(rng.uniform(20,42) if foreground else rng.uniform(12,28)),stroke_linecap='round',opacity=f(op)))
    return group(''.join(items),'FX_前景蒸汽_层' if foreground else 'FX_远景蒸汽_层')

def ik(shoulder,wrist,bend):
    sx,sy=shoulder; wx,wy=wrist; dx,dy=wx-sx,wy-sy; dist=max(1,math.hypot(dx,dy))
    l1,l2=174.,166.; d=min(dist,l1+l2-2)
    ux,uy=dx/dist,dy/dist; a=(l1*l1-l2*l2+d*d)/(2*d); h=math.sqrt(max(0,l1*l1-a*a))
    mx,my=sx+ux*a,sy+uy*a
    return (mx-uy*h*bend,my+ux*h*bend)

def water_drops(t,dry):
    rng=random.Random(SEED+301); s=[]
    # 体表水滴：垂直加速，生命周期由 y=944 的地面交点计算，末端融入水渍。
    for i in range(36):
        x=rng.uniform(825,1095); y=rng.uniform(355,680); birth=rng.uniform(-1.5,2.7); g=420
        life=math.sqrt(2*(944-y)/g); age=t-birth
        if 0<age<life:
            yy=y+.5*g*age*age; xx=x+math.sin(i)*8*age
            op=.75*smooth(age,0,.12)*(1-smooth(age,life-.12,life))
            s.append(group(ellipse(0,0,5,10,fill=P['water'],opacity='0.75')+circle(-1.5,-3,1.7,fill=P['white'],opacity='0.9'),f'FX_水珠_{i:03d}',transform=f'translate({f(xx)} {f(yy)})',opacity=f(op)))
        if life-.12<=age<life+.10:
            u=clamp((age-life+.12)/.22)
            s.append(ellipse(x,944,8+28*u,2+6*u,fill='none',stroke=P['water'],stroke_width='2',opacity=f(.38*(1-u)),id=f'FX_落地涟漪_{i:03d}'))
    # 甩水：初速度沿切线；由抛物线与地面交点求生命周期，不在半空提前消亡。
    for i in range(52):
        birth=4.58+rng.uniform(0,.72); age=t-birth
        side=-1 if i%2 else 1
        x0=960+side*rng.uniform(40,180); y0=rng.uniform(190,570)
        vx=side*rng.uniform(170,390); vy=rng.uniform(-230,-40); g=420
        life=(-vy+math.sqrt(vy*vy+2*g*(944-y0)))/g
        if 0<age<life:
            x=x0+vx*age; y=y0+vy*age+.5*g*age*age
            op=.9*(1-smooth(age,life-.15,life))
            s.append(group(path('M0 -9 Q8 2 0 10 Q-8 2 0 -9 Z',fill=P['water'])+circle(-2,-2,1.7,fill=P['white']),f'FX_甩水_{i:03d}',transform=f'translate({f(x)} {f(y)}) rotate({f(-side*25)})',opacity=f(op)))
        if life-.15<=age<life+.10:
            u=clamp((age-life+.15)/.25); xx=x0+vx*life
            s.append(ellipse(xx,944,10+34*u,2+7*u,fill='none',stroke=P['water'],stroke_width='2',opacity=f(.42*(1-u)),id=f'FX_甩水落地_{i:03d}'))
    return group(''.join(s),'FX_水珠与飞溅_层')

def render_frames(out:Path,frames,scale=1.0,keep_svg=True):
    import cairosvg
    svgdir=out/'SVG帧序列'; pngdir=out/'PNG帧序列'
    if keep_svg: svgdir.mkdir(parents=True,exist_ok=True)
    pngdir.mkdir(parents=True,exist_ok=True)
    for n in frames:
        svg=scene(n/FPS)
        if keep_svg: (svgdir/f'帧{n:03d}.svg').write_text(svg,encoding='utf-8')
        cairosvg.svg2png(bytestring=svg.encode('utf-8'),write_to=str(pngdir/f'帧{n:03d}.png'),output_width=round(W*scale),output_height=round(H*scale))
        print(f'渲染帧 {n:03d}/{max(frames):03d}',flush=True)

def contact_sheet(out:Path,name='D2_关键帧总览.png',before=False):
    from PIL import Image,ImageDraw,ImageFont
    try: font=ImageFont.truetype('/usr/share/fonts/opentype/noto/NotoSansCJK-Regular.ttc',28)
    except: font=ImageFont.load_default()
    try: small=ImageFont.truetype('/usr/share/fonts/opentype/noto/NotoSansCJK-Regular.ttc',22)
    except: small=font
    tw,th=480,270; margin=24; labelh=46; cols=3; rows=math.ceil(len(BEATS)/cols)
    sheet=Image.new('RGB',(cols*(tw+margin)+margin,rows*(th+labelh+margin)+margin),(14,27,35)); d=ImageDraw.Draw(sheet)
    for i,(n,label) in enumerate(BEATS):
        im=Image.open(out/'PNG帧序列'/f'帧{n:03d}.png').convert('RGB'); im.thumbnail((tw,th),Image.Resampling.LANCZOS)
        x=margin+(i%cols)*(tw+margin); y=margin+(i//cols)*(th+labelh+margin)
        sheet.paste(im,(x,y)); d.text((x,y+th+7),f'帧{n:03d} · {n/FPS:.2f}s · {label}',font=small,fill=(234,224,202))
    sheet.save(out/name)

def export_keyframes(out:Path):
    import shutil
    dest=out/'D2_关键帧'; dest.mkdir(parents=True,exist_ok=True)
    for n,label in BEATS:
        shutil.copy2(out/'PNG帧序列'/f'帧{n:03d}.png',dest/f'帧{n:03d}_{n/FPS:.2f}s_{label}.png')

def main():
    ap=argparse.ArgumentParser(description='渲染《浴后·暖意》全矢量动画')

    ap.add_argument('--keyframes-only',action='store_true')
    ap.add_argument('--frames',default='')
    ap.add_argument('--scale',type=float,default=1.0)
    ap.add_argument('--video',action='store_true')
    args=ap.parse_args(); out=Path(args.out); out.mkdir(parents=True,exist_ok=True)
    if args.frames: frames=[int(x) for x in args.frames.split(',')]
    elif args.keyframes_only: frames=[n for n,_ in BEATS]
    else: frames=list(range(round(DURATION*FPS)))
    render_frames(out,frames,args.scale)
    if args.keyframes_only or args.video:
        contact_sheet(out); export_keyframes(out)
    if args.video:
        subprocess.run(['ffmpeg','-y','-framerate',str(FPS),'-i',str(out/'PNG帧序列/帧%03d.png'),'-c:v','libx264','-preset','slow','-crf','18','-pix_fmt','yuv420p','-movflags','+faststart',str(out/'D3_浴后暖意_整片预览.mp4')],check=True)
